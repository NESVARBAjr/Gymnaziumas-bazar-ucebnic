        window.handleImageSelection = (event, type) => {
            const file = event.target.files[0];
            if (!file) return;
            if (!file.type.startsWith('image/')) { window.showToast("CHYBA: Nutný obrázek."); return; }

            const reader = new FileReader();
            reader.readAsDataURL(file);
            reader.onload = function(e) {
                const img = new Image();
                img.src = e.target.result;
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    // Pro QR kód zachováme vyšší rozlišení (800px), pro knihu stačí 400px
                    const MAX_WIDTH = type === 'qr' ? 800 : 400; 
                    const scaleSize = Math.min(MAX_WIDTH / img.width, 1); // Nezvětšovat malé fotky
                    canvas.width = img.width * scaleSize;
                    canvas.height = img.height * scaleSize;
                    const ctx = canvas.getContext('2d');
                    
                    // Bílý podklad (pokud by fotka měla průhledné pozadí)
                    ctx.fillStyle = "#FFFFFF";
                    ctx.fillRect(0, 0, canvas.width, canvas.height);
                    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
                    
                    // Pokud uživatel nahrává QR kód, provedeme detekci přes jsQR
                    if (type === 'qr') {
                        const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                        const code = jsQR(imageData.data, imageData.width, imageData.height, {
                            inversionAttempts: "dontInvert",
                        });

                        if (!code) {
                            window.showToast("CHYBA: V obrázku nebyl detekován žádný platný QR kód.");
                            document.getElementById('qr-upload').value = ""; // Reset file inputu
                            return; // Přerušíme nahrávání
                        } else {
                            window.showToast("Výborně, QR kód úspěšně ověřen.");
                        }
                    }

                    const compressed = canvas.toDataURL('image/jpeg', 0.8);

                    if (type === 'book') {
                        currentUploadedImageBase64 = compressed;
                        document.getElementById('image-preview').src = compressed;
                        document.getElementById('image-preview').classList.remove('hidden');
                        document.getElementById('upload-placeholder').classList.add('hidden');

                        document.getElementById('qr-preview').classList.remove('hidden');
                        document.getElementById('qr-upload-placeholder').classList.add('hidden');
                        document.getElementById('remove-qr-btn').classList.remove('hidden');
                    }
                }
            };
        };

                payment: { cash: payCash, qr: payQr ? currentUploadedQRBase64 : null },
                status: 'available',
                timestamp: Date.now()
            };

            // Uložení do lokální databáze v paměti
            window.dbListings.push(newListing);

            window.showToast('Inzerát přidán a uložen!');
            document.getElementById('add-form').reset();
            window.removeImage(null, 'book');
            window.removeImage(null, 'qr');
            window.toggleQRSection();
            window.goToStep1();
            
            activeCategory = 'Vše';
            window.renderCategories();
            window.filterListings();
            window.switchView('view-home');
        };

        // === ZOBRAZENÍ A FILTRY ===
        window.renderCategories = () => {
            const container = document.getElementById('category-filters');
            container.innerHTML = '';
            categories.forEach(cat => {
                const btn = document.createElement('button');
                btn.className = `whitespace-nowrap px-5 py-2.5 rounded-[14px] text-sm transition-all duration-200 border ${cat === activeCategory ? 'bg-bazar-dark text-white font-black border-bazar-dark' : 'bg-white text-gray-600 font-bold border-gray-200 hover:bg-gray-50 shadow-sm'}`;
                btn.innerText = cat;
                btn.onclick = () => { activeCategory = cat; window.renderCategories(); window.filterListings(); };
                container.appendChild(btn);
            });
        };

        window.filterListings = () => {
            const query = document.getElementById('search-input').value.toLowerCase();
            const filtered = window.dbListings.filter(item => {
                const isAvailable = item.status !== 'sold'; 
                const matchCategory = activeCategory === 'Vše' || item.subject === activeCategory;
                const matchSearch = item.title.toLowerCase().includes(query) || item.subject.toLowerCase().includes(query);
                return isAvailable && matchCategory && matchSearch;
            });
            // Seřazení podle nejnovějšího
            filtered.sort((a,b) => (b.timestamp || 0) - (a.timestamp || 0));
            renderHomeListings(filtered);
        };

        function renderHomeListings(dataToRender) {
            const grid = document.getElementById('listings-grid');
            const emptyState = document.getElementById('empty-state');
            grid.innerHTML = '';

            if (dataToRender.length === 0) {
                grid.style.display = 'none';
                emptyState.style.display = 'flex';
                return;
            }

            grid.style.display = 'grid';
            emptyState.style.display = 'none';

            dataToRender.forEach((item, index) => {
                const card = document.createElement('div');
                // Přidána třída animate-slide-up-fade pro kaskádovité zobrazení
                card.className = "animate-slide-up-fade bg-white rounded-[1.5rem] p-3 shadow-premium border border-gray-100 flex flex-col cursor-pointer transition-all duration-300 transform active:scale-95 hover:shadow-xl hover:-translate-y-1 relative overflow-hidden group";
                // Nastavení zpoždění animace (stagger effect)
                card.style.animationDelay = `${index * 0.08}s`;
                card.onclick = () => window.openDetail(item);

                card.innerHTML = `
                    <div class="absolute inset-0 bg-gradient-to-b from-transparent to-gray-50/50 pointer-events-none"></div>
                    <div class="relative w-full aspect-[3/4] rounded-2xl overflow-hidden mb-4 bg-bazar-gray border border-gray-100/50 shadow-inner">
                        <img src="${item.image}" class="w-full h-full object-cover transition-transform duration-500 group-hover:scale-105">
                        <div class="absolute top-2 left-2 bg-white/90 backdrop-blur-md text-[10px] font-black px-2.5 py-1.5 rounded-lg text-bazar-dark uppercase tracking-wider shadow-sm border border-white/50">${item.condition}</div>
                    </div>
                    <div class="px-1 flex flex-col flex-1 relative z-10">
                        <h3 class="font-black text-bazar-dark text-sm leading-tight line-clamp-2 mb-1">${item.title}</h3>
                        <p class="text-[11px] text-gray-500 font-bold mb-3 uppercase tracking-wider">${item.subject}</p>
                        <div class="mt-auto flex justify-between items-center pt-2">
                            <p class="font-black text-transparent bg-clip-text bg-gradient-to-r from-bazar-green to-emerald-600 text-lg">${item.price} Kč</p>
                            <div class="w-8 h-8 rounded-full bg-gray-50 border border-gray-100 flex items-center justify-center text-bazar-dark group-hover:bg-bazar-dark group-hover:text-white group-hover:border-bazar-dark transition-colors shadow-sm"><i data-lucide="arrow-right" class="w-4 h-4 stroke-[2.5]"></i></div>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
            lucide.createIcons();
        }

        // === DETAIL KNIHY A PROCES NÁKUPU ===
        window.openDetail = (item) => {
            const container = document.getElementById('detail-content');
            const isMyListing = window.currentUser && window.currentUser.email === item.sellerEmail;
            
            let actionButton = isMyListing 
                ? `<button onclick="window.switchView('view-home')" class="w-full h-14 bg-gray-100 text-gray-400 rounded-2xl font-black text-base flex justify-center items-center gap-2 cursor-not-allowed border border-gray-200">Váš inzerát</button>`
                : `<button onclick="window.openCheckout('${item.id}')" class="w-full h-14 bg-gradient-to-r from-bazar-green to-emerald-600 text-white rounded-2xl font-black text-base shadow-glow active:scale-95 transition-transform flex justify-center items-center gap-2 hover:shadow-[0_0_40px_rgba(16,185,129,0.5)]">Koupit za ${item.price} Kč</button>`;

            container.innerHTML = `
                <div class="relative h-[22rem] bg-bazar-gray cursor-pointer group" onclick="window.openGallery('${item.image}')">
                    <img src="${item.image}" class="w-full h-full object-cover">
                    
                    <!-- Indikátor fotogalerie -->
                    <div class="absolute inset-0 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity bg-black/20">
                       <div class="w-16 h-16 bg-black/50 rounded-full flex items-center justify-center text-white backdrop-blur-md shadow-lg transform transition-transform group-hover:scale-110">
                          <i data-lucide="zoom-in" class="w-8 h-8"></i>
                       </div>
                    </div>
                    
                    <div class="absolute top-0 left-0 w-full p-6 safe-pt flex justify-between items-center z-10 bg-gradient-to-b from-black/60 via-black/20 to-transparent pointer-events-none">
                        <button onclick="event.stopPropagation(); window.switchView('view-home')" class="w-12 h-12 rounded-[18px] bg-white/20 backdrop-blur-md flex items-center justify-center text-white active:scale-90 transition-transform border border-white/20 shadow-lg pointer-events-auto hover:bg-white/30"><i data-lucide="arrow-left" class="w-6 h-6"></i></button>
                    </div>
                    <div class="absolute bottom-12 right-4 bg-black/60 text-white text-xs font-bold px-3 py-1.5 rounded-full backdrop-blur-md z-10 flex items-center gap-1 pointer-events-none">
                        <i data-lucide="image" class="w-3 h-3"></i> 1/1
                    </div>
                </div>
                <div class="bg-white rounded-t-[2.5rem] -mt-10 relative px-6 pt-8 pb-32 shadow-[0_-10px_40px_rgba(0,0,0,0.05)]">
                    <div class="flex justify-between items-center mb-4">
                        <span class="px-4 py-2 bg-green-50 text-bazar-green rounded-xl text-xs font-black uppercase tracking-wider border border-green-100">${item.subject}</span>
                        <p class="text-3xl font-black text-transparent bg-clip-text bg-gradient-to-r from-bazar-green to-emerald-600">${item.price} Kč</p>
                    </div>
                    <h1 class="text-2xl font-black text-bazar-dark mb-8 leading-tight">${item.title}</h1>
                    
                    <div class="grid grid-cols-2 gap-4 mb-8">
                        <div class="bg-bazar-gray rounded-2xl p-4 flex flex-col justify-center border border-gray-100">
                            <p class="text-[11px] text-gray-500 font-bold uppercase tracking-wider mb-1">Stav</p>
                            <p class="font-black text-bazar-dark text-sm">${item.condition}</p>
                        </div>
                        <div class="bg-bazar-gray rounded-2xl p-4 flex flex-col justify-center border border-gray-100">
                            <p class="text-[11px] text-gray-500 font-bold uppercase tracking-wider mb-1">Přidáno</p>
                            <p class="font-black text-bazar-dark text-sm">${item.dateAdded}</p>
                        </div>
                    </div>

                    <div class="mb-8">
                        <h3 class="text-xs font-black text-gray-400 uppercase tracking-wider mb-3">Info od prodejce</h3>
                        <p class="text-gray-700 text-sm leading-relaxed font-medium">${item.description || 'Bez popisu.'}</p>
                    </div>

                    <div class="flex items-center justify-between p-4 bg-white border border-gray-200 rounded-[1.5rem] shadow-sm mb-4">
                        <div class="flex items-center gap-4">
                            <div class="w-12 h-12 bg-bazar-dark text-white rounded-[14px] flex items-center justify-center font-black text-lg shadow-md">${item.seller.charAt(0).toUpperCase()}</div>
                            <div>
                                <p class="text-[11px] text-gray-400 font-bold uppercase tracking-wider">Prodejce</p>
                                <p class="font-black text-bazar-dark">${item.seller}</p>
                            </div>
                        </div>
                    </div>
                    ${!isMyListing ? `<button onclick="window.reportListing('${item.id}')" class="w-full py-3 text-xs font-bold text-gray-400 text-center active:scale-90 transition-transform">Nahlásit problém s inzerátem</button>` : ''}
                </div>
                <div class="fixed bottom-0 left-0 w-full p-6 bg-white/80 backdrop-blur-2xl border-t border-gray-200/50 safe-pb sm:absolute sm:rounded-b-[44px]">
                    ${actionButton}
                </div>
            `;
            lucide.createIcons();
            window.switchView('view-detail');
        };

        window.openCheckout = (id) => {
            currentItemForCheckout = window.dbListings.find(l => l.id === id);
            const item = currentItemForCheckout;
            const canCash = item.payment ? item.payment.cash : true; 
            const qrData = item.payment ? item.payment.qr : null;

            let html = `
                <div class="bg-white rounded-[2rem] p-4 border border-gray-200 flex gap-4 mb-8 items-center shadow-sm">
                    <img src="${item.image}" class="w-20 h-24 object-cover rounded-2xl bg-bazar-gray">
                    <div class="flex-1 pr-2">
                        <h3 class="font-black text-bazar-dark text-sm leading-tight line-clamp-2 mb-1">${item.title}</h3>
                        <p class="text-xs text-gray-500 font-bold mb-2">Prodává: ${item.seller}</p>
                        <p class="text-xl font-black text-bazar-green">${item.price} Kč</p>
                    </div>
                </div>
                <h2 class="text-sm font-black text-bazar-dark mb-4 uppercase tracking-wider">Způsob platby</h2>
                <div class="space-y-3 mb-8">
            `;
            
            if (canCash) {
                html += `
                    <label class="flex items-center p-5 bg-white border border-gray-200 rounded-2xl cursor-pointer hover:bg-gray-50 focus-within:ring-2 focus-within:border-bazar-green transition shadow-sm">
                        <input type="radio" name="checkout-payment" value="cash" class="w-5 h-5 text-bazar-green border-gray-300" ${!qrData ? 'checked' : ''} onchange="window.updateCheckoutUI()">
                        <div class="ml-4 flex-1"><span class="block font-black text-bazar-dark">Hotově při předání</span></div>
                    </label>
                `;
            }
            if (qrData) {
                html += `
                    <label class="flex items-center p-5 bg-white border border-gray-200 rounded-2xl cursor-pointer hover:bg-gray-50 focus-within:ring-2 focus-within:border-bazar-green transition shadow-sm">
                        <input type="radio" name="checkout-payment" value="qr" class="w-5 h-5 text-bazar-green border-gray-300" ${!canCash || qrData ? 'checked' : ''} onchange="window.updateCheckoutUI()">
                        <div class="ml-4 flex-1"><span class="block font-black text-bazar-dark">Přes QR kód na místě</span></div>
                    </label>
                `;
            }
            html += `</div>
                <div id="checkout-qr-display" class="hidden flex-col items-center p-8 bg-bazar-gray border border-gray-200 rounded-[2rem] mb-8 text-center">
                    <p class="text-sm font-black text-bazar-dark mb-4">Naskenujete při osobním předání s prodejcem</p>
                    <div class="p-3 bg-white rounded-2xl shadow-sm border border-gray-200 inline-block"><img src="${qrData}" class="w-48 h-48 object-contain rounded-xl" onclick="window.openGallery('${qrData}')"></div>
                </div>
                <button onclick="window.confirmPurchase()" class="w-full h-14 bg-bazar-dark text-white rounded-2xl font-black text-base shadow-xl active:scale-95 transition flex justify-center items-center gap-2 mt-4">
                    Potvrdit závazný nákup <i data-lucide="check" class="w-5 h-5"></i>
                </button>
            `;
            document.getElementById('checkout-content').innerHTML = html;
            lucide.createIcons();
            window.switchView('view-checkout');
            window.updateCheckoutUI();
        };

        window.updateCheckoutUI = () => {
            const selected = document.querySelector('input[name="checkout-payment"]:checked');
            const qrDisplay = document.getElementById('checkout-qr-display');
            if (selected && selected.value === 'qr' && currentItemForCheckout.payment.qr) {
                qrDisplay.classList.remove('hidden');
                qrDisplay.classList.add('flex');
            } else if (qrDisplay) {
                qrDisplay.classList.add('hidden');
                qrDisplay.classList.remove('flex');
            }
        };

        window.confirmPurchase = () => {
            const selected = document.querySelector('input[name="checkout-payment"]:checked');
            if(!selected) { window.showToast("CHYBA: Vyberte způsob platby."); return; }
            if (!window.currentUser) return;
            
            // Označení inzerátu jako "prodaný" v lokální databázi
            const listing = window.dbListings.find(l => l.id === currentItemForCheckout.id);
            if(listing) {
                listing.status = 'sold';
                listing.buyerEmail = window.currentUser.email; // Zapíšeme, kdo to koupil
            }
            
            window.showToast("Nákup úspěšný! Inzerát zarezervován.");
            
            // Okamžité překreslení dat
            window.filterListings();
            setTimeout(() => { window.switchView('view-home'); }, 1500);
        };

        window.showToast = (message) => {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            const isError = message.includes("CHYBA");
            
            toast.className = `${isError ? "bg-red-500 text-white" : "bg-bazar-dark text-white shadow-glow"} px-6 py-4 rounded-2xl shadow-2xl text-sm font-bold flex items-center justify-center text-center transform -translate-y-10 opacity-0 transition-all duration-300 border border-white/10 z-[2000] relative`;
            toast.innerText = message.replace("CHYBA:", "");
            
            container.appendChild(toast);
            
            requestAnimationFrame(() => {
                toast.classList.remove('-translate-y-10', 'opacity-0');
                toast.classList.add('translate-y-0', 'opacity-100');
            });
            
            setTimeout(() => { 
                toast.classList.remove('translate-y-0', 'opacity-100'); 
                toast.classList.add('-translate-y-10', 'opacity-0'); 
                setTimeout(() => toast.remove(), 300); 
            }, 3500);
        };

        window.onload = () => { lucide.createIcons(); };
    </script>
